cmake_minimum_required(VERSION 3.22)
project(PlaybackNativeEngine VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)

add_library(PlaybackEngineCore STATIC engine/EngineCore.cpp engine/StreamingSource.cpp engine/MixerRouter.cpp engine/SampleRateConverter.cpp engine/DeviceBackend.cpp engine/PlaybackSources.cpp)
target_include_directories(PlaybackEngineCore PUBLIC "${CMAKE_CURRENT_LIST_DIR}/engine")
target_compile_features(PlaybackEngineCore PUBLIC cxx_std_20)

if(BUILD_TESTING)
  add_executable(PlaybackEngineCoreTests engine/tests/EngineCoreTests.cpp)
  target_link_libraries(PlaybackEngineCoreTests PRIVATE PlaybackEngineCore)
  add_test(NAME PlaybackEngineCoreTests COMMAND PlaybackEngineCoreTests)
endif()

add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../external/JUCE" JUCE)
add_library(PlaybackEngineJuceBackend STATIC device/JuceDeviceBackend.cpp)
target_include_directories(PlaybackEngineJuceBackend PUBLIC "${CMAKE_CURRENT_LIST_DIR}/device")
target_compile_features(PlaybackEngineJuceBackend PUBLIC cxx_std_20)
target_link_libraries(PlaybackEngineJuceBackend PUBLIC PlaybackEngineCore juce::juce_audio_utils juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
target_compile_definitions(PlaybackEngineJuceBackend PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_ASIO=1)
juce_add_console_app(PlaybackEngineDeviceTest PRODUCT_NAME "Playback Engine Device Test")
target_sources(PlaybackEngineDeviceTest PRIVATE device/DeviceTestMain.cpp)
target_compile_features(PlaybackEngineDeviceTest PRIVATE cxx_std_20)
target_link_libraries(PlaybackEngineDeviceTest PRIVATE PlaybackEngineJuceBackend juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
juce_add_console_app(PlaybackEngineV3Test PRODUCT_NAME "Playback Engine V3 Test")
target_sources(PlaybackEngineV3Test PRIVATE device/ManifestTestMain.cpp)
target_compile_features(PlaybackEngineV3Test PRIVATE cxx_std_20)
target_link_libraries(PlaybackEngineV3Test PRIVATE PlaybackEngineJuceBackend juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
juce_add_console_app(PlaybackEngineProbe PRODUCT_NAME "Playback Engine Probe")
target_sources(PlaybackEngineProbe PRIVATE src/Main.cpp)
target_compile_features(PlaybackEngineProbe PRIVATE cxx_std_20)
target_compile_definitions(PlaybackEngineProbe PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_ASIO=1)
target_link_libraries(PlaybackEngineProbe PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
add_executable(PlaybackNativeAsioDiagnostic device/NativeAsioDiagnosticMain.cpp)
target_compile_features(PlaybackNativeAsioDiagnostic PRIVATE cxx_std_20)
target_include_directories(PlaybackNativeAsioDiagnostic PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../external/JUCE/modules/juce_audio_devices/native/asio")
target_link_libraries(PlaybackNativeAsioDiagnostic PRIVATE ole32)
